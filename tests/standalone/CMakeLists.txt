cmake_minimum_required(VERSION 3.20)
project(RE8DisplayStandaloneTests LANGUAGES CXX)
enable_testing()
foreach(name IN ITEMS geometry projection cursor gui_layout invoke_numeric viewport media_layout method_lookup prompt_layout)
    add_executable(re8_${name}_tests ${name}.cpp)
    target_include_directories(re8_${name}_tests PRIVATE ../../native/common)
    target_compile_features(re8_${name}_tests PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(re8_${name}_tests PRIVATE /EHsc /UNDEBUG)
    else()
        target_compile_options(re8_${name}_tests PRIVATE -UNDEBUG)
    endif()
    add_test(NAME re8_${name}_tests COMMAND re8_${name}_tests)
endforeach()
add_executable(re8_plugin_abi_tests plugin_abi.cpp)
target_include_directories(re8_plugin_abi_tests PRIVATE ../../external/reframework-sdk)
target_compile_features(re8_plugin_abi_tests PRIVATE cxx_std_20)
target_compile_definitions(re8_plugin_abi_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
    target_compile_options(re8_plugin_abi_tests PRIVATE /EHsc /UNDEBUG)
endif()
add_test(NAME re8_plugin_abi_tests COMMAND re8_plugin_abi_tests "${CMAKE_CURRENT_SOURCE_DIR}/../../artifacts/native/Release/RE8Display.dll")
